function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow(['ID','First name','Last name','Category','Amount (CAD)','Expense date','Pay period','Cutoff date','Description','Submitted']); } const data = JSON.parse(e.postData.contents); sheet.appendRow([ data.id, data.firstName, data.lastName, data.category, data.amount, data.expenseDate, data.payPeriod, data.cutoffDate, data.description, data.submitted ]); return ContentService .createTextOutput(JSON.stringify({ success: true, id: data.id })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ success: false, error: err.message })) .setMimeType(ContentService.MimeType.JSON); } }
